const apiUrl = '/api/gemini'; async function askGemini() { const text = chatInput.value.trim(); if (!text || sendBtn.disabled) return; addMessage(text, 'user'); chatInput.value = ''; sendBtn.disabled = true; const typing = addTyping(); try { const response = await fetch(apiUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: text, stage: document.querySelector('.node.active')?.dataset.index ?? 0 }) }); if (!response.ok) { throw new Error('Backend unavailable'); } const data = await response.json(); typing.remove(); addMessage(data.answer || 'No answer returned.', 'bot'); statusNote.textContent = 'Live Gemini response is active via secure Vercel function.'; } catch (error) { typing.remove(); addMessage(getFallbackAnswer(text), 'bot'); statusNote.textContent = 'Fallback demo mode active. Add GEMINI_API_KEY in Vercel to enable the live Gemini API.'; } finally { sendBtn.disabled = false; chatInput.focus(); } } sendBtn.addEventListener('click', askGemini); chatInput.addEventListener('keydown', (event) => { if (event.key === 'Enter') askGemini(); });
